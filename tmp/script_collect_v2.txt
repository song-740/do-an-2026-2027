function Log {
    param([string]$msg, [string]$level = 'INFO')
    $ts = Get-Date -Format "HH:mm:ss.fff"
    $line = "[$ts][$level] $msg"
    $color = switch ($level) {
        'ERROR' { 'Red' }
        'WARN'  { 'Yellow' }
        'OK'    { 'Green' }
        default { 'Cyan' }
    }
    Write-Host $line -ForegroundColor $color
    $line | Out-File "$OutRoot\_collection.log" -Append -Encoding UTF8
}

function Timed {
    param([string]$Name, [scriptblock]$Block)
    $t0 = Get-Date
    Log "=== $Name ==="
    try {
        & $Block
        $status = 'OK'
    } catch {
        $errDetail = "$($_.Exception.Message)`n$($_.ScriptStackTrace)`n$($_.Exception.StackTrace)"
        Log "  [$Name] EXCEPTION: $($_.Exception.Message)" 'ERROR'
        $errDetail | Out-File "$OutRoot\00_meta\_exceptions.txt" -Append -Encoding UTF8
        $status = "FAIL: $($_.Exception.Message)"
    }
    $t1 = Get-Date
    $dur = [Math]::Round(($t1 - $t0).TotalSeconds, 2)
    "[$Name] $($t0.ToString('HH:mm:ss.fff')) -> $($t1.ToString('HH:mm:ss.fff')) | ${dur}s | $status" |
        Out-File "$OutRoot\00_meta\_timing.txt" -Append -Encoding UTF8
    Log "  -> $Name done in ${dur}s ($status)" $(if ($status -eq 'OK') {'OK'} else {'ERROR'})
}

function SafeCmd {
    param([string]$Cmd, [string]$OutFile)
    try {
        $output = & cmd.exe /c $Cmd 2>&1
        $output | Out-File $OutFile -Encoding UTF8
    } catch {
        "ERROR: $($_.Exception.Message)" | Out-File $OutFile -Encoding UTF8
    }
}

function Copy-LockedFile {
    # Fallback chain: RawCopy -> robocopy /B -> Copy-Item
    # Có log lý do fail từng phương án
    param(
        [string]$Source,
        [string]$DestDir,
        [string]$LogFile = "$OutRoot\00_meta\_copy_locked_failures.txt"
    )
    $dest = Join-Path $DestDir (Split-Path $Source -Leaf)
    $reasons = @()

    # 1. RawCopy
    $rc = Join-Path $ToolsDir "RawCopy64.exe"
    if (Test-Path $rc) {
        & $rc /FileNamePath:$Source /OutputPath:$DestDir 2>&1 | Out-Null
        if (Test-Path $dest) { return $true }
        $reasons += "RawCopy: file not created"
    } else {
        $reasons += "RawCopy: tool missing"
    }

    # 2. robocopy /B
    robocopy (Split-Path $Source) $DestDir (Split-Path $Source -Leaf) /B /R:0 /W:0 /NJH /NJS 2>&1 | Out-Null
    if (Test-Path $dest) { return $true }
    $reasons += "robocopy /B: exit $LASTEXITCODE"

    # 3. Copy-Item
    try {
        Copy-Item $Source $DestDir -Force -EA Stop
        return $true
    } catch {
        $reasons += "Copy-Item: $($_.Exception.Message)"
    }

    "$Source | FAILED | $($reasons -join ' || ')" |
        Out-File $LogFile -Append -Encoding UTF8
    return $false
}

function Get-FileMetadataHash {
    param([string]$Path)
    try {
        $h = (Get-FileHash $Path -Algorithm SHA256 -EA Stop).Hash
        return $h
    } catch {
        return "ERROR: $($_.Exception.Message)"
    }
}

# DỌN ORPHAN HIVE TỪ LẦN CHẠY TRƯỚC (nếu script crash giữa chừng)
# ============================================================
try {
    $hkuList = reg query HKU 2>&1
    foreach ($line in $hkuList) {
        if ($line -match 'TempHive_') {
            $orphanKey = ($line -split '\s+')[0]
            if ($orphanKey -match 'TempHive_') {
                Log "Cleaning orphan hive from previous run: $orphanKey" 'WARN'
                reg unload $orphanKey 2>&1 | Out-Null
            }
        }
    }
} catch {
    Log "Orphan hive cleanup skipped: $($_.Exception.Message)" 'WARN'
}

            Signature = if ($sig) { $sig.Status } else { 'N/A' }
            Signer    = if ($sig) { $sig.SignerCertificate.Subject } else { 'N/A' }
        }
    }
    $toolReport | Format-List | Out-File "$m\tool_integrity.txt" -Encoding UTF8

# BƯỚC 2: RAM DUMP (BLOCKING)
# ============================================================
Timed "01_ram_dump_BLOCKING" {
    $wmem = Join-Path $ToolsDir "winpmem_mini_x64.exe"
    if (-not (Test-Path $wmem)) {
        Log "  [!] winpmem_mini_x64.exe không tìm thấy — BỎ QUA RAM DUMP" 'WARN'
        "MISSING: winpmem_mini_x64.exe" | Out-File "$OutRoot\01_volatile\_ram_dump_status.txt" -Encoding UTF8
        return
    }

    # Check disk space trước khi dump (cần ít nhất = RAM size + buffer)
    $ramGB  = [Math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB, 2)
    $drive  = $OutRoot.Substring(0,2)
    $freeGB = [Math]::Round((Get-PSDrive -Name $OutRoot.Substring(0,1)).Free / 1GB, 2)
    $evidSizeGB = [Math]::Round(((Get-ChildItem $OutRoot -Recurse -File |
        Where-Object { $excludeNames -notcontains $_.Name } |
        Measure-Object Length -Sum).Sum / 1GB), 2)

    if ($freeGB -lt ($evidSizeGB + 1)) {
        Log "  [!] Không đủ dung lượng để zip (cần ~$($evidSizeGB+1) GB, còn $freeGB GB)" 'WARN'
        Log "  Bỏ qua nén — chuyển thư mục $OutRoot trực tiếp" 'WARN'
        return   # skip zip
}
    Log "  RAM: ${ramGB} GB | Free on $drive : ${freeGB} GB"

    if ($freeGB -lt ($ramGB + 2)) {
        Log "  [!] INSUFFICIENT DISK SPACE (need ~$($ramGB+2) GB, have ${freeGB} GB)" 'ERROR'
        "INSUFFICIENT SPACE: need $($ramGB+2) GB, have $freeGB GB" |
            Out-File "$OutRoot\01_volatile\_ram_dump_status.txt" -Encoding UTF8
        return
    }

    $ramOut = "$OutRoot\01_volatile\memory.raw"
    Log "  RAM dump starting (BLOCKING) — có thể mất 5-30 phút..."
    $proc = Start-Process $wmem -ArgumentList $ramOut -PassThru -WindowStyle Hidden -Wait

    if (Test-Path $ramOut) {
        $size = (Get-Item $ramOut).Length
        $hash = Get-FileMetadataHash $ramOut
        @(
            "RAM dump status: SUCCESS"
            "Path: $ramOut"
            "Size bytes: $size"
            "Size GB: $([Math]::Round($size/1GB, 3))"
            "SHA256: $hash"
            "Start: $($StartTime.ToString('yyyy-MM-dd HH:mm:ss'))"
            "End: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
        ) | Out-File "$OutRoot\01_volatile\_ram_dump_status.txt" -Encoding UTF8
        Log "  RAM dump OK: $([Math]::Round($size/1GB,2)) GB, SHA256=$($hash.Substring(0,16))..." 'OK'
    } else {
        "FAILED: file not created" | Out-File "$OutRoot\01_volatile\_ram_dump_status.txt" -Encoding UTF8
        Log "  RAM dump FAILED" 'ERROR'
    }

    # Pagefile / hiberfil / swapfile
    foreach ($f in @('C:\pagefile.sys','C:\hiberfil.sys','C:\swapfile.sys')) {
        if (Test-Path $f) {
            Log "  Copying $f..."
            Copy-LockedFile -Source $f -DestDir "$OutRoot\01_volatile" | Out-Null
        }
    }
}

Timed "03_processes" {
    $p = "$OutRoot\03_processes"

    Get-Process -EA SilentlyContinue | Select-Object `
        Id, ProcessName, StartTime, Company, Description, Path,
        @{N='CPU';E={$_.CPU}},
        @{N='WS_MB';E={[Math]::Round($_.WorkingSet64/1MB,2)}},
        @{N='SessionId';E={$_.SessionId}} |
        Sort-Object StartTime | Out-File "$p\process_details.txt" -Encoding UTF8

    Get-CimInstance Win32_Process -EA SilentlyContinue | ForEach-Object {
        $owner = try { $_.GetOwner().User } catch { 'N/A' }
        [PSCustomObject]@{
            PID         = $_.ProcessId
            PPID        = $_.ParentProcessId
            Name        = $_.Name
            CommandLine = $_.CommandLine
            ExecutablePath = $_.ExecutablePath
            CreationDate = $_.CreationDate
            Owner       = $owner
            SessionId   = $_.SessionId
        }
    } | Sort-Object PID | Export-Csv "$p\process_list_full.csv" -NoTypeInformation -Encoding UTF8

    Get-CimInstance Win32_Process -EA SilentlyContinue |
        Select-Object @{N='PID';E={$_.ProcessId}},
                      @{N='PPID';E={$_.ParentProcessId}},
                      @{N='Name';E={$_.Name}},
                      @{N='CMD';E={$_.CommandLine}},
                      @{N='Path';E={$_.ExecutablePath}} |
        Sort-Object PPID,PID | Format-Table -AutoSize -Wrap |
        Out-File "$p\process_tree.txt" -Encoding UTF8

    SafeCmd "tasklist /v" "$p\tasklist_verbose.txt"
    SafeCmd "tasklist /svc" "$p\tasklist_services.txt"

    Get-CimInstance Win32_Process -EA SilentlyContinue |
        Where-Object { $_.ExecutablePath } | ForEach-Object {
            $h = try { (Get-FileHash $_.ExecutablePath -Algorithm SHA256 -EA Stop).Hash }
                 catch { "ERROR: $($_.Exception.Message)" }
            "$($_.ProcessId) | $($_.Name) | $h | $($_.ExecutablePath)"
        } | Out-File "$p\process_hashes.txt" -Encoding UTF8

    $lldll = Join-Path $ToolsDir "Listdlls64.exe"
    if (Test-Path $lldll) {
        & $lldll -accepteula 2>&1 | Out-File "$p\loaded_dlls_all.txt" -Encoding UTF8
    }
}

# BƯỚC 9: EVENT LOGS
# ============================================================
Timed "08_eventlogs" {
    $e = "$OutRoot\08_eventlogs"

    $channels = Get-WinEvent -ListLog * -EA SilentlyContinue
    Log "  Total channels: $($channels.Count)"

    $channels | Select-Object LogName, RecordCount, LogMode, MaximumSizeInBytes,
        IsEnabled, LogFilePath, FileSize, LastWriteTime, LastAccessTime |
        Sort-Object LogName | Export-Csv "$e\channel_inventory.csv" -NoTypeInformation -Encoding UTF8

    $emptyReport = @()
    foreach ($ch in $channels) {
        $logName = $ch.LogName
        $safe = $logName -replace '[\\/:*?"<>|]', '_'

        if ($ch.RecordCount -gt 0) {
            $evtx = "$e\$safe.evtx"
            wevtutil epl "$logName" "$evtx" 2>&1 | Out-Null
        }

        if ($ch.RecordCount -eq 0 -or -not $ch.IsEnabled) {
            $meta = wevtutil gl "$logName" 2>&1 | Out-String
            $emptyReport += [PSCustomObject]@{
                LogName      = $logName
                RecordCount  = $ch.RecordCount
                IsEnabled    = $ch.IsEnabled
                FileSize     = $ch.FileSize
                LastWrite    = $ch.LastWriteTime
                Metadata     = $meta
            }
        }
    }
    $emptyReport | Export-Csv "$e\_empty_or_disabled_channels.csv" -NoTypeInformation -Encoding UTF8

    $secIds = @(1102, 4624, 4625, 4648, 4672, 4688, 4698, 4699, 4700, 4702)
    foreach ($id in $secIds) {
        try {
            Get-WinEvent -FilterHashtable @{LogName='Security'; Id=$id} -MaxEvents 2000 -EA Stop |
                Select-Object TimeCreated, Id, ProviderName, Message |
                Out-File "$e\security_id_$id.txt" -Encoding UTF8
        } catch {}
    }

    $sysmonIds = @(1,3,7,8,10,11,12,13,14,17,18,19,20,21,22,23,25)
    foreach ($id in $sysmonIds) {
        try {
            Get-WinEvent -FilterHashtable @{LogName='Microsoft-Windows-Sysmon/Operational'; Id=$id} -MaxEvents 5000 -EA Stop |
                Select-Object TimeCreated, Id, Message |
                Out-File "$e\sysmon_id_$id.txt" -Encoding UTF8
        } catch {}
    }
}

# BƯỚC 10: REGISTRY HIVES
# ============================================================
Timed "09_registry" {
    $r = "$OutRoot\09_registry"
    $raw = "$r\raw_hives"
    New-Item -ItemType Directory -Path $raw -Force | Out-Null

    $hives = @(
        @{Name='HKLM_SOFTWARE'; Path='HKLM\SOFTWARE'}
        @{Name='HKLM_SYSTEM';   Path='HKLM\SYSTEM'}
        @{Name='HKLM_SECURITY'; Path='HKLM\SECURITY'}
        @{Name='HKLM_SAM';      Path='HKLM\SAM'}
        @{Name='HKCU_ALL';      Path='HKCU'}
    )
    foreach ($h in $hives) {
        reg export $h.Path "$r\$($h.Name).reg" /y 2>&1 | Out-Null
        if ($LASTEXITCODE -ne 0) {
            "FAILED (exit $LASTEXITCODE): $($h.Path)" | Out-File "$r\_export_errors.txt" -Append
        }
    }

    $rawHives = @(
        "$env:SystemRoot\System32\config\SAM"
        "$env:SystemRoot\System32\config\SYSTEM"
        "$env:SystemRoot\System32\config\SOFTWARE"
        "$env:SystemRoot\System32\config\SECURITY"
        "$env:SystemRoot\System32\config\DEFAULT"
    )
    foreach ($hv in $rawHives) {
        Copy-LockedFile -Source $hv -DestDir $raw | Out-Null
    }

    # NTUSER.DAT + UsrClass.dat — có try/finally + GC trước unload
    Get-ChildItem "C:\Users" -Directory -EA SilentlyContinue | ForEach-Object {
        $uname = $_.Name
        $ntu = Join-Path $_.FullName "NTUSER.DAT"
        $cls = Join-Path $_.FullName "AppData\Local\Microsoft\Windows\UsrClass.dat"

        if (Test-Path $ntu) {
            $ok = Copy-LockedFile -Source $ntu -DestDir $raw
            if (-not $ok) {
                $key = "HKU\TempHive_$uname"
                # Dọn orphan nếu có từ lần trước
                reg unload $key 2>&1 | Out-Null

                reg load $key $ntu 2>&1 | Out-Null
                if ($LASTEXITCODE -eq 0) {
                    try {
                        reg export $key "$raw\NTUSER_$uname.reg" /y 2>&1 | Out-Null
                        if ($LASTEXITCODE -ne 0) {
                            "reg export failed for $uname (exit $LASTEXITCODE)" |
                                Out-File "$raw\_reg_export_errors.txt" -Append
                        }
                    } finally {
                        # GC trước unload — PS giữ handle đến registry key
                        [gc]::Collect()
                        [gc]::WaitForPendingFinalizers()
                        [gc]::Collect()
                        reg unload $key 2>&1 | Out-Null
                    }
                } else {
                    "reg load failed for $uname (exit $LASTEXITCODE)" |
                        Out-File "$raw\_reg_load_errors.txt" -Append
                }
            }
        }
        if (Test-Path $cls) {
            Copy-LockedFile -Source $cls -DestDir $raw | Out-Null
        }
    }

    $queries = @(
        'HKCU\Software\Classes'
        'HKLM\SOFTWARE\Classes'

        # ------------------------------------------------------------
        Log "  Exporting USN Journal via fsutil..."
        $usnCsv = "$ntfs\usnjrnl_fsutil.csv"
        fsutil usn readjournal C: csv 2>&1 | Out-File $usnCsv -Encoding UTF8
        if ($LASTEXITCODE -ne 0 -or -not (Test-Path $usnCsv) -or (Get-Item $usnCsv).Length -eq 0) {
            Log "  fsutil failed, trying ExtractUsnJrnl..." 'WARN'
            $extract = Join-Path $ToolsDir "ExtractUsnJrnl.exe"

                & $extract -f "C:\`$Extend\`$UsnJrnl" -o $ntfs 2>&1 |
                    Out-File "$ntfs\_extractusn_log.txt" -Encoding UTF8
            } else {
                # Fallback: RawCopy chỉ để lấy raw $J (phân tích sau bằng tool khác)
                Log "  ExtractUsnJrnl missing, trying RawCopy raw \$J..." 'WARN'
                & (Join-Path $ToolsDir "RawCopy64.exe") `
                    /FileNamePath:"C:\`$Extend\`$UsnJrnl:`$J" /OutputPath:$ntfs 2>&1 |
                    Out-File "$ntfs\_rawcopy_usn_log.txt" -Encoding UTF8

# BƯỚC 16: ANTI-FORENSIC INDICATORS
# ============================================================
Timed "15_anti_forensic" {
    $a = "$OutRoot\16_anti_forensic"
    $report = @()

    # 1. Channel bị disable
    $channels = Get-WinEvent -ListLog * -EA SilentlyContinue |
        Where-Object { -not $_.IsEnabled }
    if ($channels) {
        $report += "=== DISABLED CHANNELS ==="
        $channels | ForEach-Object { $report += "  $($_.LogName) (LastWrite: $($_.LastWriteTime))" }
        $channels | Select-Object LogName, LastWriteTime, FileSize |
            Export-Csv "$a\disabled_channels.csv" -NoTypeInformation -Encoding UTF8
    }

    # 2. Channel bị clear
    $cleared = Get-WinEvent -ListLog * -EA SilentlyContinue |
        Where-Object { $_.RecordCount -eq 0 -and $_.IsEnabled -and $_.FileSize -lt 70000 -and $_.LogFilePath }
    if ($cleared) {
        $report += ""
        $report += "=== SUSPICIOUS EMPTY CHANNELS (possible clear) ==="
        $cleared | ForEach-Object {
            $report += "  $($_.LogName) | FileSize=$($_.FileSize) | LastWrite=$($_.LastWriteTime)"
        }
        $cleared | Select-Object LogName, FileSize, LastWriteTime |
            Export-Csv "$a\suspicious_empty_channels.csv" -NoTypeInformation -Encoding UTF8
    }

    # 3. Security event 1102
    try {
        $ev1102 = Get-WinEvent -FilterHashtable @{LogName='Security'; Id=1102} -MaxEvents 100 -EA Stop
        if ($ev1102) {
            $report += ""
            $report += "=== SECURITY LOG CLEARED (Event 1102) ==="
            $ev1102 | ForEach-Object { $report += "  $($_.TimeCreated) | $($_.Message)" }
        }
    } catch {}

    # 4. Registry policy tắt logging
    $policyKeys = @(
        'HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\ScriptBlockLogging'
        'HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\ModuleLogging'
        'HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\Transcription'
        'HKLM:\SOFTWARE\Wow6432Node\Policies\Microsoft\Windows\PowerShell\ScriptBlockLogging'
        'HKLM:\SOFTWARE\Wow6432Node\Policies\Microsoft\Windows\PowerShell\ModuleLogging'
        'HKLM:\SOFTWARE\Wow6432Node\Policies\Microsoft\Windows\PowerShell\Transcription'
    )
    $report += ""
    $report += "=== POWERSHELL LOGGING POLICY ==="
    foreach ($k in $policyKeys) {
        if (Test-Path $k) {
            $props = Get-ItemProperty $k -EA SilentlyContinue
            $report += "  $k :"
            $props.PSObject.Properties | Where-Object { $_.Name -notlike 'PS*' } |
                ForEach-Object { $report += "    $($_.Name) = $($_.Value)" }
        }
    }

    # 5. Scheduled task có action wevtutil cl / vssadmin delete
    $tasks = Get-ScheduledTask -EA SilentlyContinue
    $suspiciousTasks = $tasks | Where-Object {
        $task = $_
        ($task.Actions | Where-Object {
            $_.Execute -match 'cmd|powershell' -and
            $_.Arguments -match 'wevtutil\s+cl|vssadmin\s+delete|Clear-EventLog'
        } | Measure-Object).Count -gt 0
    }
    if ($suspiciousTasks) {
        $report += ""
        $report += "=== SCHEDULED TASKS WITH LOG CLEARING ==="
        $suspiciousTasks | ForEach-Object {
            $report += "  $($_.TaskPath)$($_.TaskName)"
            $_.Actions | ForEach-Object { $report += "    Action: $($_.Execute) $($_.Arguments)" }
        }
        $suspiciousTasks | Select-Object TaskName, TaskPath |
            Export-Csv "$a\suspicious_tasks_logclear.csv" -NoTypeInformation -Encoding UTF8
    }

    # 6. Shadow copy bị xóa
    $shadowOut = & vssadmin list shadows 2>&1 | Out-String
    if ($shadowOut -match 'No items found|Không tìm thấy') {
        $report += ""
        $report += "=== NO SHADOW COPIES (possibly deleted) ==="
    }

    # 7. Defender bị tắt
    $defStatus = Get-MpComputerStatus -EA SilentlyContinue
    if ($defStatus) {
        $report += ""
        $report += "=== DEFENDER STATUS ==="
        $report += "  RealTimeProtectionEnabled: $($defStatus.RealTimeProtectionEnabled)"
        $report += "  AntivirusEnabled: $($defStatus.AntivirusEnabled)"
        $report += "  TamperProtectionSource: $($defStatus.TamperProtectionSource)"
    }

    # 8. Process cha-con bất thường
    $suspParent = Get-CimInstance Win32_Process -EA SilentlyContinue |
        Where-Object { $_.Name -match '^(powershell|pwsh|cmd|wscript|cscript|mshta|rundll32|regsvr32)\.exe$' } |
        ForEach-Object {
            $parent = Get-CimInstance Win32_Process -Filter "ProcessId=$($_.ParentProcessId)" -EA SilentlyContinue
            if ($parent -and $parent.Name -notmatch '^(explorer|cmd|powershell|pwsh|services|svchost|wininit|userinit|taskeng|taskhostw|schtasks|wmiprvse)\.exe$') {
                [PSCustomObject]@{
                    PID = $_.ProcessId
                    Name = $_.Name
                    CMD = $_.CommandLine
                    PPID = $_.ParentProcessId
                    ParentName = $parent.Name
                    ParentCMD = $parent.CommandLine
                }
            }
        }
    if ($suspParent) {
        $report += ""
        $report += "=== SUSPICIOUS PROCESS PARENT-CHILD ==="
        $suspParent | ForEach-Object {
            $report += "  $($_.Name) (PID $($_.PID)) <- parent $($_.ParentName) (PPID $($_.PPID))"
            $report += "    CMD: $($_.CMD)"
        }
        $suspParent | Export-Csv "$a\suspicious_parent_child.csv" -NoTypeInformation -Encoding UTF8
    }

    $report | Out-File "$a\_ANTI_FORENSIC_INDICATORS.txt" -Encoding UTF8
    Log "  Anti-forensic report written ($($report.Count) lines)"
}

Hostname         : $env:COMPUTERNAME
Collected by     : $env:USERNAME
Is Admin         : $IsAdmin
Script version   : v2.1

CẤU TRÚC THƯ MỤC

00_meta         : Thời gian, OS/hardware/user, hash & signature tools, timing, exceptions

03_processes    : Process list (PID/PPID/CMD/owner), process tree, hash executable, DLLs

----------------------

HẠN CHẾ
-------

        "Evidence Collection - $CaseID"
        "Collected by  : $env:USERNAME @ $env:COMPUTERNAME"

        "Collection End  : $($EndTime.ToString('yyyy-MM-dd HH:mm:ss zzz'))"

    Log "============================================================" 'OK'
